Protect backend API routes dynamically by intercepting incoming requests, evaluating rules against user sessions and request payloads, and returning structured denial reasons.

---

## Express Middleware Guard

```ts
import { Request, Response, NextFunction } from "express";
import { Rule, checkRule } from "@qarar/sdk";

/**
 * Creates an Express middleware that checks incoming requests against a Qarar rule.
 */
export function createRuleGuard(rule: Rule) {
  return (req: Request, res: Response, next: NextFunction) => {
    // Construct data context combining user session and request payload
    const context = {
      user: (req as any).user,
      body: req.body,
      query: req.query,
      params: req.params,
    };

    const check = checkRule(rule, context);

    if (!check.result) {
      return res.status(403).json({
        error: "Forbidden",
        message: "Request failed business policy verification",
        reasons: check.reasons,
      });
    }

    next();
  };
}
```

## Route Usage Example

```ts
import express from "express";
import { rule } from "@qarar/sdk";
import { createRuleGuard } from "./ruleGuard";

const app = express();
app.use(express.json());

// 1. Define authorization policy
const adminOnlyWith2FA = rule("Admin 2FA Verification")
  .where("user.role").eq("admin")
  .and("user.hasMfaEnabled").eq(true)
  .build();

// 2. Attach middleware to sensitive endpoint
app.post(
  "/api/system/purge-cache",
  createRuleGuard(adminOnlyWith2FA),
  (req, res) => {
    res.json({ status: "Cache purged successfully" });
  }
);
```
